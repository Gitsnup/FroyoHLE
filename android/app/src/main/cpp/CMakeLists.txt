cmake_minimum_required(VERSION 3.22.1)
project(froyohle_android)

add_library(froyohle SHARED native_bridge.cpp)
find_library(log-lib log)
find_library(gles1-lib GLESv1_CM)
set(rust_core "${CMAKE_CURRENT_LIST_DIR}/prebuilt/${ANDROID_ABI}/libfroyohle.a")
if(EXISTS "${rust_core}")
    add_library(froyohle_core STATIC IMPORTED)
    set_target_properties(froyohle_core PROPERTIES IMPORTED_LOCATION "${rust_core}")
    target_link_libraries(froyohle ${log-lib} ${gles1-lib} froyohle_core android log z dl m)
else()
    target_compile_definitions(froyohle PRIVATE FROYOHLE_NO_CORE=1)
    target_link_libraries(froyohle ${log-lib} ${gles1-lib})
endif()
